cmake_minimum_required(VERSION 3.16)
project(tire_health_service LANGUAGES CXX)
set(THS_FUNCTIONAL_PROFILE "v1" CACHE STRING "Accepted Tire product content profile (v1 only)")
if(NOT THS_FUNCTIONAL_PROFILE STREQUAL "v1")
    message(FATAL_ERROR "Tire supports only THS_FUNCTIONAL_PROFILE=v1; release versions are separate")
endif()
find_package(Threads REQUIRED)
add_library(tire_health_runtime src/runtime/json.cpp src/runtime/sha256.cpp src/runtime/uuid.cpp src/runtime/protocol.cpp src/runtime/posix.cpp src/runtime/application.cpp src/runtime/token_session.cpp src/model.cpp src/state.cpp src/runtime.cpp)
target_include_directories(tire_health_runtime PUBLIC include)
target_compile_features(tire_health_runtime PUBLIC cxx_std_17)
target_link_libraries(tire_health_runtime PUBLIC Threads::Threads)
set_target_properties(tire_health_runtime PROPERTIES CXX_EXTENSIONS OFF CXX_STANDARD_REQUIRED ON)
target_compile_options(tire_health_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(tire-health-bootstrap src/runtime/bootstrap_main.cpp)
target_link_libraries(tire-health-bootstrap PRIVATE tire_health_runtime)
option(THS_BUILD_KUKSA_RUNTIME "Build real pinned TLS KUKSA product executable" OFF)
if(THS_BUILD_KUKSA_RUNTIME)
  include(cmake/KuksaRuntime.cmake)
else()
  message(STATUS "Host/domain targets only; product executable NOT built")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(native_inputs_tests tests/native_inputs_tests.cpp)
    target_link_libraries(native_inputs_tests PRIVATE tire_health_runtime)
    target_compile_options(native_inputs_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME native_service_inputs COMMAND native_inputs_tests)
    add_executable(tire_token_session_tests tests/token_session_tests.cpp)
    target_link_libraries(tire_token_session_tests PRIVATE tire_health_runtime)
    target_compile_options(tire_token_session_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME tire_private_token_session COMMAND tire_token_session_tests)
  add_executable(tire_health_tests tests/core_tests.cpp)
  target_link_libraries(tire_health_tests PRIVATE tire_health_runtime)
  # Assertions include contract checks and must execute in product Release builds.
  target_compile_options(tire_health_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  add_test(NAME tire_health_contract COMMAND tire_health_tests)
endif()
